  function renderFindings(){
    const gs=selectedRecords(),f=OpeningRoastLive.findings(gs,liveReport.ownImport?.games||[]),s=stats(gs),esc=escapeHTML;
    const warning=[...liveReport.imported.warnings,liveReport.ownWarning].filter(Boolean).join(' ');$('liveWarnings').textContent=warning;
    $('adjustedResults').textContent=s.adjustedN?'Rating-adjusted result: '+(s.adjusted*100).toFixed(1)+' percentage points vs Elo expectation across '+s.adjustedN+' games.':'No comparable rating data.';
    $('liveRatings').innerHTML=Object.entries(liveReport.imported.ratings).filter(([k,v])=>k.startsWith('chess_')&&v.last?.rating).map(([k,v])=>'<span>'+esc(k.replace('chess_',''))+' <b>'+v.last.rating+'</b></span>').join('');
    $('liveTheory').textContent='Average matched opening prefix: '+f.avgBook.toFixed(1)+' half-moves across '+gs.length+' games. This measures overlap with the opening database, not memorization.';
    $('livePhases').innerHTML=Object.entries(f.phases).map(([name,p])=>'<article class="paper-card"><span class="tag">'+esc(name)+'</span><h3>'+p.n+' player moves</h3><p>Mean expected-score loss: '+(100*p.loss/p.n).toFixed(1)+' percentage points. '+p.errors+' mistakes or blunders.</p><p class="muted">'+(p.n<20?'Small sample; no reliable phase conclusion.':'Descriptive sample measurement.')+'</p></article>').join('')||'<p>No analyzed moves match these filters.</p>';
    const evidence=(g,m,text)=>'<button class="text-button" data-evidence="'+esc(g.id)+'" data-ply="'+m.ply+'">'+esc(text)+' →</button>';
    $('liveCritical').innerHTML=f.critical.map(({game:g,move:m})=>'<article class="paper-card weakness"><span class="tag">'+esc(m.label)+' · '+esc(m.phase)+'</span><h3>'+esc(g.opponent)+' · '+g.date+'</h3><p>'+Math.ceil(m.ply/2)+(m.ply%2?'. ':'... ')+esc(g.notation[m.ply-1])+': '+(m.loss*100).toFixed(1)+' points of expected-score loss.</p>'+evidence(g,m,'Review actual game')+'</article>').join('')||'<p>No mistakes found at this search budget.</p>';
    $('prepPersonal').textContent=liveReport.ownImport?liveReport.ownImport.user+': '+liveReport.ownImport.games.length+' recent games imported. '+f.overlap+' of '+gs.length+' opponent games share an opening classification with your sample. Your games provide repertoire comparison; only the opponent’s games receive full engine analysis in this version.':liveReport.ownWarning||'Add your own username to compare opening repertoires.';
    const prep=f.recurring.length?f.recurring.map(r=>({...r,recurring:true})):f.critical.slice(0,3).map(r=>({...r,count:1,recurring:false}));
    $('livePrep').innerHTML=prep.map(({game:g,move:m,count,recurring},i)=>'<article class="prep-item"><span class="prep-number">0'+(i+1)+'</span><div><span class="tag">'+(recurring?count+' occurrences of the same position':'One-game study example · limited evidence')+'</span><h3>'+esc(g.opening)+'</h3><p>Before '+esc(g.notation[m.ply-1])+' the engine suggests:</p><div class="variation-line">'+esc(OpeningRoastLive.sanLine(g.fens[m.ply-1],m.pv))+'</div>'+evidence(g,{ply:m.ply-1},'Study position and alternative')+'</div></article>').join('')||'<p>No supported preparation examples in these filters.</p>';
    $('downloadReport').onclick=()=>{const blob=new Blob([JSON.stringify({engine:OpeningRoastLive.VERSION,fetchedAt:liveReport.imported.fetchedAt,player,own,modes:included,summary:s,games:gs.map(({positions,...g})=>g)},null,2)],{type:'application/json'});const a=document.createElement('a');a.href=URL.createObjectURL(blob);a.download='opening-roast-'+player+'.json';a.click();setTimeout(()=>URL.revokeObjectURL(a.href),1000);};
  }
